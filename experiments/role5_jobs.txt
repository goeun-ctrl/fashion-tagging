[P2b] --exp_id R5_cnxt_muon_x4_e10_s42 --optimizer muon --muon_lr_scale 4 --epochs 10 --seed 42 --num_workers 6

[P4] --exp_id R5_eva_adamw_s42 --model eva02_small_patch14_336.mim_in22k_ft_in1k --set_img_size --size 378 --layer_decay 0.75 --drop_path 0.1 --seed 42

[P4] --exp_id R5_eva_adamw_s43 --model eva02_small_patch14_336.mim_in22k_ft_in1k --set_img_size --size 378 --layer_decay 0.75 --drop_path 0.1 --seed 43
[P4] --exp_id R5_eva_muon_s43 --model eva02_small_patch14_336.mim_in22k_ft_in1k --set_img_size --size 378 --layer_decay 0.75 --drop_path 0.1 --optimizer muon --seed 43
